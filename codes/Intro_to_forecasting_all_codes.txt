## Code 6 ##

#MLR

import statsmodels.api as sm

# Define dependent and independent variables
y = df["Cement_Sales_000_Tons"]

X = df[
    [
        "Advertising_Rs_Lakh",
        "Housing_Starts_000_Units",
        "Average_Cement_Price_Rs_Bag",
        "Rainfall_mm",
        "GDP_Growth_Percent"
    ]
]

# Add intercept
X = sm.add_constant(X)

# Fit MLR model
model = sm.OLS(y, X).fit()

# Display regression results
print(model.summary())

## End ## 

## Code 7 ##

#Time Series

df["Quarter"] = pd.PeriodIndex(df["Quarter"], freq="Q")
plt.figure(figsize=(10, 5))

plt.plot(df["Quarter"].astype(str), df["Cement_Sales_000_Tons"], marker="o")

plt.title("Quarterly Cement Sales")
plt.xlabel("Quarter")
plt.ylabel("Cement Sales (000 Tons)")

plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
